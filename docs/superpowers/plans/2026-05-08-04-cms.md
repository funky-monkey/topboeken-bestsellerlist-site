# TopBoeken — Plan 4: Express CMS Admin Panel

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the password-protected `/admin` CMS — dashboard, book editor, source manager, affiliate manager, scrape log viewer, manual scrape trigger, and "Publiceer nu" rebuild button.

**Architecture:** Express 4 server with `express-session` for auth. Plain HTML views (no templating framework — just template literal functions). All routes are server-rendered. Session stored in SQLite. Runs on port 3001 behind Nginx.

**Tech Stack:** Express 4, express-session, bcrypt, better-sqlite3

**Prerequisite:** Plan 1 complete.

```js
// server/index.js
import express from 'express';
import session from 'express-session';
import { getDb } from '../src/db/db.js';

import authRoutes      from './routes/auth.js';
import dashboardRoutes from './routes/dashboard.js';
import booksRoutes     from './routes/books.js';
import sourcesRoutes   from './routes/sources.js';
import affiliatesRoutes from './routes/affiliates.js';
import logsRoutes      from './routes/logs.js';
import scrapeRoutes    from './routes/scrape.js';
import rebuildRoutes   from './routes/rebuild.js';
import { requireAuth } from './middleware/auth.js';

const app = express();
const PORT = process.env.PORT ?? 3001;

app.use(express.urlencoded({ extended: true }));
app.use(express.json());

app.use(session({
  secret: process.env.SESSION_SECRET ?? 'change-me-in-production',
  resave: false,
  saveUninitialized: false,
  cookie: { maxAge: 24 * 60 * 60 * 1000, httpOnly: true, sameSite: 'strict' },
}));

// Auth routes (no requireAuth)
app.use('/admin', authRoutes);

// Protected routes
app.use('/admin',    requireAuth, dashboardRoutes);
app.use('/admin',    requireAuth, booksRoutes);
app.use('/admin',    requireAuth, sourcesRoutes);
app.use('/admin',    requireAuth, affiliatesRoutes);
app.use('/admin',    requireAuth, logsRoutes);
app.use('/admin',    requireAuth, scrapeRoutes);
app.use('/api',      requireAuth, rebuildRoutes);

app.listen(PORT, () => console.log(`CMS running on http://localhost:${PORT}`));
```

```js
// server/routes/auth.js
import { Router } from 'express';
import bcrypt from 'bcrypt';
import { loginView } from '../views/login.js';

const router = Router();

router.get('/login', (req, res) => {
  if (req.session?.authenticated) return res.redirect('/admin');
  res.send(loginView());
});

router.post('/login', async (req, res) => {
  const { password } = req.body;
  const hash = process.env.ADMIN_PASSWORD_HASH;
  if (!hash) return res.send(loginView('ADMIN_PASSWORD_HASH is not set in environment.'));

  const ok = await bcrypt.compare(password, hash);
  if (!ok) return res.send(loginView('Ongeldig wachtwoord.'));

  req.session.authenticated = true;
  res.redirect('/admin');
});

router.post('/logout', (req, res) => {
  req.session.destroy(() => res.redirect('/admin/login'));
});

// GET logout for convenience
router.get('/logout', (req, res) => {
  req.session.destroy(() => res.redirect('/admin/login'));
});

export default router;
```

```js
// server/routes/books.js
import { Router } from 'express';
import { getDb } from '../../src/db/db.js';
import { booksListView, bookEditView } from '../views/books.js';

const router = Router();

router.get('/books', (req, res) => {
  const q = req.query.q ?? '';
  const db = getDb();
  const books = q
    ? db.prepare('SELECT * FROM books WHERE title LIKE ? OR author LIKE ? ORDER BY updated_at DESC LIMIT 100').all(`%${q}%`, `%${q}%`)
    : db.prepare('SELECT * FROM books ORDER BY updated_at DESC LIMIT 100').all();
  res.send(booksListView(books, q));
});

router.get('/books/:id', (req, res) => {
  const db = getDb();
  const book = db.prepare('SELECT * FROM books WHERE id = ?').get(req.params.id);
  if (!book) return res.status(404).send('Boek niet gevonden');
  const genres     = db.prepare('SELECT * FROM genres ORDER BY name_nl').all();
  const bookGenres = db.prepare('SELECT g.* FROM book_genres bg JOIN genres g ON g.id=bg.genre_id WHERE bg.book_id=?').all(book.id);
  const flash = req.query.saved ? 'Wijzigingen opgeslagen.' : null;
  res.send(bookEditView(book, genres, bookGenres, flash));
});

router.post('/books/:id', (req, res) => {
  const db  = getDb();
  const { title, author, summary, cover_path, genres } = req.body;
  const id  = parseInt(req.params.id, 10);

  db.prepare(`
    UPDATE books SET title=?, author=?, summary=?, cover_path=?, updated_at=datetime('now') WHERE id=?
  `).run(title, author, summary || null, cover_path || null, id);

  // Update genres
  db.prepare('DELETE FROM book_genres WHERE book_id=?').run(id);
  const insertGenre = db.prepare('INSERT OR IGNORE INTO book_genres (book_id, genre_id) VALUES (?,?)');
  const genreIds = Array.isArray(genres) ? genres : genres ? [genres] : [];
  db.transaction(() => genreIds.forEach(gid => insertGenre.run(id, parseInt(gid, 10))))();

  res.redirect(`/admin/books/${id}?saved=1`);
});

export default router;
```

```js
// server/routes/scrape.js
import { Router } from 'express';
import { spawn } from 'node:child_process';
import { join, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
import { layout } from '../views/layout.js';

const __dirname = dirname(fileURLToPath(import.meta.url));
const router = Router();

router.post('/scrape', (req, res) => {
  const scriptPath = join(__dirname, '../../scripts/scrape.sh');
  // Run in background — don't await
  const child = spawn('bash', [scriptPath], {
    detached: true,
    stdio: 'ignore',
  });
  child.unref();
  res.send(layout('Scrapen gestart', `
    <h1>Scrapen gestart</h1>
    <p>De scraper draait op de achtergrond. Bekijk de voortgang in de <a href="/admin/logs">logs</a>.</p>
    <p style="margin-top:16px"><a href="/admin" class="btn btn-primary">Terug naar dashboard</a></p>
  `));
});

export default router;
```
